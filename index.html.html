Таануу коду (Python)

# реле, сенсор жана жүздөр базасы
lock = OutputDevice(17)
pir  = MotionSensor(4)
db   = np.load("faces.npz")      # names[], enc[N,128]
THRESHOLD = 0.45

def check_face():
    frame = capture_rgb(cam)
    boxes = face_locations(frame, model="hog")
    for enc in face_encodings(frame, boxes):
        d = face_distance(db["enc"], enc)
        i = int(np.argmin(d))
        if d[i] < THRESHOLD and is_live(cam):   # көз ирмөө
            return db["names"][i], float(d[i])
    return None, None

while True:
    pir.wait_for_motion()
    name, dist = check_face()
    publish("lab502/door/event", name, dist)
    if name:
        lock.on(); time.sleep(3); lock.off()

👁 Жандуулукту текшерүү

is_live() 1–2 секунд кадр алып, EAR (көздүн ачыктыгы) эсептейт. Фото ирмебейт, ошондуктан EAR түшпөйт жана текшерүү өтпөйт.

⚡ Pi 4'төгү ылдамдык

Кадр 320×240 өлчөмүнө кичирейтилип, HOG менен табылат. Бир таанууга болжол менен 0.5–1 сек.
